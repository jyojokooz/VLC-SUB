# Cache the loaded model so we don't reload it for every chunk
_offline_model_cache = {"model": None, "model_size": None}

def get_offline_model_dir():
    """Get the directory where offline whisper models are stored."""
    model_dir = os.path.join(os.path.expanduser("~"), ".cache", "universalsubtitles", "models")
    os.makedirs(model_dir, exist_ok=True)
    return model_dir

def check_offline_available():
    """Check if faster-whisper is installed."""
    try:
        import faster_whisper
        return True
    except ImportError:
        return False

def transcribe_offline(chunk_path, model_size="medium", source_lang=None, status_callback=None, progress_callback=None):
    """
    Transcribe audio using faster-whisper (local, offline).
    Auto-detects NVIDIA GPU and uses CUDA if available, otherwise CPU.
    Returns data in the same format as the API response for compatibility.
    """
    try:
        from faster_whisper import WhisperModel
    except ImportError:
        raise Exception(
            "Offline transcription requires 'faster-whisper'.\n\n"
            "Please install it:\n"
            "  pip install faster-whisper\n\n"
            "For GPU acceleration, also install CUDA toolkit."
        )

    # Load or reuse cached model
    if _offline_model_cache["model"] is None or _offline_model_cache["model_size"] != model_size:
        model_dir = get_offline_model_dir()
        
        # --- Monkey Patch tqdm to get download progress ---
        original_tqdm = None
        try:
            import tqdm.auto
            original_tqdm = tqdm.auto.tqdm
            
            class CustomTqdm(original_tqdm):
                def update(self, n=1):
                    super().update(n)
                    if self.total and progress_callback and status_callback:
                        pct = int((self.n / self.total) * 100)
                        # Ensure we don't spam updates for small changes, just general progress
                        if hasattr(self, '_last_pct') and self._last_pct == pct:
                            return
                        self._last_pct = pct
                        mb_done = int(self.n / (1024 * 1024))
                        mb_total = int(self.total / (1024 * 1024))
                        status_callback(f"Downloading model... {pct}% ({mb_done}MB / {mb_total}MB)")
                        progress_callback(pct, 100, "%")
            
            tqdm.auto.tqdm = CustomTqdm
        except Exception:
            pass # Ignore patching errors

        # Attempt to load with CUDA (GPU), fallback to CPU if no compatible GPU
        try:
            if status_callback:
                status_callback(f"Trying to load '{model_size}' model on GPU...")
            model = WhisperModel(
                model_size,
                device="cuda",
                compute_type="float16",
                download_root=model_dir
            )
            if status_callback:
                status_callback("🚀 Loaded on GPU successfully!")
        except Exception:
            if status_callback:
                status_callback(f"GPU not available or failed. Loading '{model_size}' on CPU...")
            model = WhisperModel(
                model_size,
                device="cpu",
                compute_type="int8",
                download_root=model_dir
            )
        
        # --- Restore tqdm ---
        if original_tqdm:
            try:
                tqdm.auto.tqdm = original_tqdm
            except: pass

        _offline_model_cache["model"] = model
        _offline_model_cache["model_size"] = model_size
    else:
        model = _offline_model_cache["model"]

    # Transcribe
    transcribe_kwargs = {
        "beam_size": 5,
        "vad_filter": True,  # Voice Activity Detection — skips silence, much faster
        "vad_parameters": {"min_silence_duration_ms": 500}
    }
    if source_lang:
        transcribe_kwargs["language"] = source_lang

    segments_gen, info = model.transcribe(chunk_path, **transcribe_kwargs)

    # Convert to the same format as the API response
    segments_list = []
    for seg in segments_gen:
        text = seg.text.strip()
        if text:
            segments_list.append({
                "start": seg.start,
                "end": seg.end,
                "text": text
            })

    return {"segments": segments_list}

def download_offline_model(model_size, status_callback=None, progress_callback=None):
    """Download an offline model with real progress and cancel support."""
    if check_offline_model_exists(model_size):
        if status_callback: status_callback(f"Model '{model_size}' is already downloaded.")
        if progress_callback: progress_callback(100, 100, "%")
        return True

    model_dir = get_offline_model_dir()
    repo_id = f"Systran/faster-whisper-{model_size}"

    try:
        from huggingface_hub import HfApi, hf_hub_download
    except ImportError:
        raise Exception("huggingface_hub is not installed.")

    if status_callback: status_callback(f"Fetching file list for '{model_size}' model...")
    if progress_callback: progress_callback(0, 100, "%")

    # Get list of files in the model repo
    api = HfApi()
    repo_files = api.list_repo_files(repo_id)

    total_files = len(repo_files)
    
    import tqdm.auto
    import inspect
    import huggingface_hub.file_download
    
    class DownloadTqdm(tqdm.auto.tqdm):
        current_i = 0
        total_files_cnt = 1
        current_filename = ""
        
        def __init__(self, *args, **kwargs):
            super().__init__(*args, **kwargs)
            self._last_pct = -1
            
        def update(self, n=1):
            super().update(n)
            if self.total and progress_callback and status_callback:
                file_pct = self.n / self.total
                overall_pct = int(((self.__class__.current_i + file_pct) / self.__class__.total_files_cnt) * 100)
                
                if getattr(config, 'CANCEL_DOWNLOAD', False):
                    raise Exception("Download cancelled by user.")
                    
                if self._last_pct == overall_pct:
                    return
                self._last_pct = overall_pct
                
                mb_done = int(self.n / (1024 * 1024))
                mb_total = int(self.total / (1024 * 1024))
                
                status_callback(f"Downloading file {self.__class__.current_i+1}/{self.__class__.total_files_cnt}: {self.__class__.current_filename} ({mb_done}MB / {mb_total}MB)")
                progress_callback(overall_pct, 100, "%")

    for i, filename in enumerate(repo_files):
        # Check cancel between each file
        if getattr(config, 'CANCEL_DOWNLOAD', False):
            raise Exception("Download cancelled by user.")

        DownloadTqdm.current_i = i
        DownloadTqdm.total_files_cnt = total_files
        DownloadTqdm.current_filename = filename

        pct = int((i / total_files) * 100)
        if status_callback: status_callback(f"Downloading file {i+1}/{total_files}: {filename}")
        if progress_callback: progress_callback(pct, 100, "%")

        kwargs = {}
        if 'tqdm_class' in inspect.signature(hf_hub_download).parameters:
            kwargs['tqdm_class'] = DownloadTqdm

        original_tqdm_fd = getattr(huggingface_hub.file_download, 'tqdm', None)
        if original_tqdm_fd is not None:
            setattr(huggingface_hub.file_download, 'tqdm', DownloadTqdm)

        try:
            hf_hub_download(
                repo_id=repo_id,
                filename=filename,
                cache_dir=model_dir,
                **kwargs
            )
        finally:
            if original_tqdm_fd is not None:
                setattr(huggingface_hub.file_download, 'tqdm', original_tqdm_fd)

    if status_callback: status_callback("✅ Model downloaded successfully!")
    if progress_callback: progress_callback(100, 100, "%")
    return True


def delete_offline_model(model_size):
    """Delete the downloaded model to save space."""
    import shutil
    model_dir = get_offline_model_dir()
    prefix = f"models--Systran--faster-whisper-{model_size}"
    deleted = False
    
    if os.path.exists(model_dir):
        for item in os.listdir(model_dir):
            if item == prefix or item.endswith(f"faster-whisper-{model_size}"):
                path = os.path.join(model_dir, item)
                try:
                    if os.path.isdir(path):
                        shutil.rmtree(path)
                    else:
                        os.remove(path)
                    deleted = True
                except Exception:
                    pass
                
    # Also clear cache if it was loaded
    if _offline_model_cache["model_size"] == model_size:
        _offline_model_cache["model"] = None
        _offline_model_cache["model_size"] = None
        
    return deleted

def transcribe_audio_chunk(chunk_path, provider, api_key, source_lang=None):
    url   = ("https://api.groq.com/openai/v1/audio/transcriptions"
              if "Groq" in provider else "https://api.openai.com/v1/audio/transcriptions")
    model = "whisper-large-v3" if "Groq" in provider else "whisper-1"

    data = {"model": model, "response_format": "verbose_json"}
    if source_lang:
        data["language"] = source_lang

    max_retries = 4
    for attempt in range(max_retries):
        try:
            with open(chunk_path, "rb") as f:
                response = requests.post(
                    url, headers={"Authorization": f"Bearer {api_key}"},
                    files={"file": (os.path.basename(chunk_path), f, "audio/mp4")},
                    data=data, timeout=300
                )

            if response.status_code == 200:
                return response.json()
            elif response.status_code == 429:
                # Rate limited — wait with exponential backoff and retry
                wait_time = min(30, (2 ** attempt) * 5)  # 5s, 10s, 20s, 30s
                if attempt < max_retries - 1:
                    time.sleep(wait_time)
                    continue
                else:
                    raise Exception(
                        f"{provider} API Rate Limit: You've hit the usage quota.\n\n"
                        "💡 Solutions:\n"
                        "• Wait a few minutes and try again\n"
                        "• Switch to 'Offline (Local)' mode — no limits!\n"
                        "• Check your Groq/OpenAI plan & billing"
                    )
            elif response.status_code == 401:
                raise Exception(f"{provider} API Error: Invalid API Key. Please verify your credentials.")
            else:
                raise Exception(f"API Error {response.status_code}:\n{response.text[:400]}")

        except requests.exceptions.Timeout:
            if attempt < max_retries - 1:
                time.sleep(3)
                continue
            raise Exception(f"{provider} API timed out after multiple attempts. Try 'Offline (Local)' mode for large files.")
        except requests.exceptions.ConnectionError:
            if attempt < max_retries - 1:
                time.sleep(3)
                continue
            raise Exception("Network connection error. Check your internet connection or use 'Offline (Local)' mode.")

    # Should not reach here, but just in case
    raise Exception(f"Failed to transcribe after {max_retries} attempts.")

def split_text_for_translation(text, max_chars=3000):
    if len(text) <= max_chars: return [text]
    sentences = re.split(r'(?<=[.!?])\s+', text)
    chunks, current = [], ""
    for sentence in sentences:
        if len(current) + len(sentence) + 1 <= max_chars:
            current = (current + " " + sentence).strip()
        else:
            if current: chunks.append(current)
            if len(sentence) > max_chars:
                for i in range(0, len(sentence), max_chars):
                    chunks.append(sentence[i:i + max_chars])
                current = ""
            else:
                current = sentence
    if current: chunks.append(current)
    return chunks if chunks else [text]

def translate_single_chunk(text, target_code, romanize=False):
    """Translate a single text chunk with exponential backoff retry."""
    max_retries = 4
    for attempt in range(max_retries):
        try:
            if not romanize:
                res = GoogleTranslator(source='auto', target=target_code).translate(text)
                if res: return res
            else:
                url    = "https://translate.googleapis.com/translate_a/single"
                params = {"client": "gtx", "sl": "auto", "tl": target_code,
                          "dt": ["t", "rm"], "q": text}
                resp   = requests.get(url, params=params, timeout=15)
                if resp.status_code == 200:
                    data     = resp.json()
                    romanized = "".join(p[2] for p in data[0] if len(p) > 2 and p[2])
                    normal    = "".join(p[0] for p in data[0] if p[0])
                    return romanized.strip() or normal.strip() or text
                elif resp.status_code == 429:
                    # Google rate limited — wait longer
                    wait_time = min(30, (2 ** attempt) * 2)  # 2s, 4s, 8s, 16s
                    time.sleep(wait_time)
                    continue
        except Exception:
            wait_time = min(15, (2 ** attempt) * 1.5)  # 1.5s, 3s, 6s, 12s
            time.sleep(wait_time)
    return text  # Return original if all retries fail

def translate_text(eng_text, target_code, romanize=False):
    if target_code == "en" or not eng_text.strip():
        return eng_text
    chunks = split_text_for_translation(eng_text, config.MAX_CHARS_PER_CHUNK)
    return " ".join(translate_single_chunk(c, target_code, romanize) for c in chunks)

def translate_batch(segments, target_code, romanize=False, batch_size=None):
    """
    Translate subtitle segments in batches for 10x fewer API calls.
    Uses concurrent.futures for lightning fast parallel translation.
    
    Returns the segments list with translated text in-place.
    """
    if target_code == "en":
        return segments

    if batch_size is None:
        batch_size = config.TRANSLATION_BATCH_SIZE

    SEPARATOR = " ||| "
    total = len(segments)
    
    import concurrent.futures

    def process_batch(batch_start):
        batch_end = min(batch_start + batch_size, total)
        batch = segments[batch_start:batch_end]
        combined = SEPARATOR.join(seg["text"] for seg in batch)
        translated = translate_single_chunk(combined, target_code, romanize)
        
        result_batch = []
        if translated and SEPARATOR in translated:
            parts = translated.split(SEPARATOR)
            if len(parts) == len(batch):
                for i, part in enumerate(parts):
                    result_batch.append((batch_start + i, part.strip()))
                return result_batch
                
        # Fallback: translate individually if the separator gets mangled
        for i, seg in enumerate(batch):
            result_batch.append((batch_start + i, translate_single_chunk(seg["text"], target_code, romanize)))
        return result_batch

    with concurrent.futures.ThreadPoolExecutor(max_workers=config.MAX_TRANSLATION_WORKERS) as executor:
        futures = [executor.submit(process_batch, start) for start in range(0, total, batch_size)]
        for future in concurrent.futures.as_completed(futures):
            for idx, text in future.result():
                segments[idx]["text"] = text

    return segments

def detect_sync_offset(video_path, srt_path, status_callback=None):
    segments = parse_srt(srt_path)
    if not segments: return None, "No subtitles found in file."

    first_sub_time = segments[0]['start']
    if status_callback:
        status_callback("Analyzing audio for auto-sync (may take ~30s)...")

    try:
        cf  = subprocess.CREATE_NO_WINDOW if os.name == 'nt' else 0
        
        # Use our new foolproof path finder
        ffmpeg_path = get_ffmpeg_path()
        
        cmd = [
            ffmpeg_path, "-i", video_path, "-t", "300",
            "-af", "silencedetect=noise=-35dB:d=0.3", "-f", "null", "-"
        ]
        
        try:
            result = subprocess.run(cmd, stderr=subprocess.PIPE, stdout=subprocess.PIPE,
                                    creationflags=cf, timeout=120)
        except FileNotFoundError:
            return None, "ffmpeg.exe is missing! Please make sure ffmpeg.exe is in the application folder."
            
        stderr = result.stderr.decode('utf-8', errors='ignore')

        silence_ends = [float(x) for x in re.findall(r'silence_end: ([\d.]+)', stderr)]
        first_speech = silence_ends[0] if silence_ends else 0.0

        offset = round(first_speech - first_sub_time, 2)
        info   = (f"First speech at {first_speech:.2f}s  |  "
                  f"First subtitle at {first_sub_time:.2f}s  |  "
                  f"Suggested offset: {offset:+.2f}s")
        return offset, info
    except subprocess.TimeoutExpired:
        return None, "Audio analysis timed out."
    except Exception as e:
        return None, f"Auto-detect failed: {e}"

def fetch_online_subtitle(video_path, target_lang="en", status_callback=None):
    """
    Fetch an online subtitle using subliminal (hash based).
    Returns path to downloaded SRT if successful, None otherwise.
    """
    try:
        from subliminal import download_best_subtitles, region, scan_video
        from babelfish import Language
    except ImportError:
        raise Exception("Subliminal is not installed. Run 'pip install subliminal'")
        
    if status_callback: status_callback("Scanning video hash...")
    
    # Configure cache (only once; ignore if already configured)
    cache_dir = os.path.join(os.path.expanduser("~"), ".cache", "universalsubtitles", "subliminal")
    os.makedirs(cache_dir, exist_ok=True)
    try:
        region.configure('dogpile.cache.dbm', arguments={'filename': os.path.join(cache_dir, 'cache.dbm')})
    except Exception:
        pass  # already configured
    
    video = scan_video(video_path)
    if not video:
        raise Exception("Could not parse video metadata for hash.")
        
    if status_callback: status_callback(f"Searching for '{target_lang}' subtitles online...")
    
    # babelfish needs 3-letter ISO-639-3 codes (e.g. "eng"), not 2-letter (e.g. "en")
    # Use fromalpha2() for 2-letter codes, fall back to direct constructor for 3-letter
    try:
        lang = Language.fromalpha2(target_lang)
    except Exception:
        try:
            lang = Language(target_lang)
        except Exception:
            lang = Language('eng')  # ultimate fallback
    
    langs = {lang}
    
    # Try downloading best subtitles
    subtitles = download_best_subtitles([video], langs)
    
    if video in subtitles and subtitles[video]:
        best_sub = subtitles[video][0]
        if status_callback: status_callback("Saving subtitle...")
        
        # Write the subtitle content directly to our output path
        base = os.path.splitext(os.path.basename(video_path))[0]
        out_path = os.path.join(config.SUB_DIR, f"{base}_{target_lang}_online.srt")
        
        # Get the raw subtitle content and write it ourselves
        content = best_sub.content
        if content:
            if isinstance(content, bytes):
                content = content.decode('utf-8', errors='replace')
            with open(out_path, 'w', encoding='utf-8') as f:
                f.write(content)
            if status_callback: status_callback(f"✅ Subtitle downloaded instantly!")
            return out_path
        
        # Fallback: use subliminal's save mechanism
        from subliminal.core import save_subtitles
        save_subtitles(video, [best_sub], single=True, directory=config.SUB_DIR, encoding='utf-8')
        
        # Find the saved file
        video_base = os.path.splitext(os.path.basename(video.name))[0]
        for candidate in [
            os.path.join(config.SUB_DIR, f"{video_base}.srt"),
            os.path.join(config.SUB_DIR, f"{video_base}.{target_lang}.srt"),
        ]:
            if os.path.exists(candidate):
                import shutil
                shutil.move(candidate, out_path)
                if status_callback: status_callback(f"✅ Subtitle downloaded instantly!")
                return out_path
        
        return None
    else:
        raise Exception(f"No '{target_lang}' subtitle found online for this video.\nTip: Subtitles are matched by file hash — try a well-known movie or TV show.")

    romanize         = gen_romanize_var.get()
    source_lang_name = gen_source_lang_var.get()
    source_lang_code = config.WHISPER_LANGUAGES.get(source_lang_name)
    is_offline       = "Offline" in provider
    offline_model_name_ui = offline_model_var.get()

    if not video_path: return messagebox.showerror("Error", "Please select a video file!")
    if not os.path.exists(video_path): return messagebox.showerror("Error", "Video file not found!")

    # API key only required for cloud providers
    if not is_offline:
        if not api_key: return messagebox.showerror("Error", "Please enter your API Key!")
        save_api_key(api_key)

    def _task():
        try:
            app.after(0, lambda: gen_btn.config(state=tk.DISABLED))
            if os.path.exists(config.TEMP_DIR):
                for fn in os.listdir(config.TEMP_DIR):
                    try: os.remove(os.path.join(config.TEMP_DIR, fn))
                    except: pass

            update_status("Compressing audio for AI (Wait ~15s)...")
            chunk_pattern = os.path.join(config.TEMP_DIR, "chunk_%03d.m4a")
            cf  = subprocess.CREATE_NO_WINDOW if os.name == 'nt' else 0
            
            ffmpeg_path = get_ffmpeg_path()
            segment_time = str(config.AUDIO_SEGMENT_TIME)
            cmd = [ffmpeg_path, "-y", "-i", video_path, "-vn", "-c:a", "aac", "-b:a", "64k",
                   "-ar", "16000", "-ac", "1", "-af", "loudnorm", "-f", "segment",
                   "-segment_time", segment_time, chunk_pattern]

            try:
                proc = subprocess.run(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, creationflags=cf)
                if proc.returncode != 0:
                    raise Exception(f"FFmpeg failed. Is the file a valid video?\n\n{proc.stderr.decode('utf-8', errors='ignore')[-400:]}")
            except FileNotFoundError:
                raise Exception("ffmpeg.exe not found! Make sure ffmpeg.exe is in the app folder.")

            chunks = sorted([f for f in os.listdir(config.TEMP_DIR) if f.endswith('.m4a')])
            if not chunks:
                raise Exception("No audio chunks extracted — the video may have no audio track.")

            all_segments = []
            for i, chunk in enumerate(chunks):
                lang_info = f" ({source_lang_name})" if source_lang_code else " (Auto-Detect)"

                if is_offline:
                    # Offline: use faster-whisper locally
                    model_name = config.OFFLINE_MODEL_SIZES.get(offline_model_name_ui, "medium")
                    update_status(f"🖥️ Transcribing Part {i+1}/{len(chunks)} locally{lang_info}...")
                    update_progress(i, len(chunks), "parts")
                    ai_data = transcribe_offline(
                        os.path.join(config.TEMP_DIR, chunk),
                        model_size=model_name,
                        source_lang=source_lang_code,
                        status_callback=update_status,
                        progress_callback=update_progress
                    )
                else:
                    # Cloud API: Groq or OpenAI
                    update_status(f"☁️ Transcribing Part {i+1}/{len(chunks)}{lang_info}...")
                    update_progress(i, len(chunks), "parts")
                    ai_data = transcribe_audio_chunk(
                        os.path.join(config.TEMP_DIR, chunk), provider, api_key, source_lang_code
                    )

                offset = i * config.AUDIO_SEGMENT_TIME
                for seg in ai_data.get("segments", []):
                    text = seg["text"].strip()
                    if text:
                        all_segments.append({"start": seg["start"] + offset, "end": seg["end"] + offset, "text": text})

            if not all_segments: raise Exception("No speech detected.")

            target_code = config.LANGUAGES.get(target_lang, "en")
            if target_code != "en":
                mode = "Manglish/Hinglish" if romanize else "Native Script"
                update_status(f"🌐 Batch translating {len(all_segments)} lines ({mode})...")
                update_progress(0, 100, "%")
                translate_batch(all_segments, target_code, romanize)
                update_progress(100, 100, "%")

            srt_content = "".join(f"{i+1}\n{format_timestamp(s['start'])} --> {format_timestamp(s['end'])}\n{s['text']}\n\n"
                                  for i, s in enumerate(all_segments) if s['text'])
            suffix   = f"_{target_lang}_Romanized" if romanize else f"_{target_lang}"
            srt_name = f"{os.path.splitext(os.path.basename(video_path))[0]}{suffix}.srt"
            srt_path = clean_path(os.path.join(config.SUB_DIR, srt_name))

            with open(srt_path, 'w', encoding='utf-8') as f: f.write(srt_content)
            for fn in os.listdir(config.TEMP_DIR):
                try: os.remove(os.path.join(config.TEMP_DIR, fn))
                except: pass

            update_status(f"✅ Done! {len(all_segments)} subtitle lines generated.")
            update_progress(100, 100, "%")
            populate_all_tabs(srt_path)
            
            # Apply subtitle to VLC
            apply_subtitle_to_vlc(app, srt_path, f"✅ Subtitles Generated!\n{len(all_segments)} lines | {target_lang}", video_path)

        except Exception as e:
            app.after(0, lambda: messagebox.showerror("Generation Error", str(e)))
            update_status("Error occurred.")
        finally:
            app.after(0, lambda: gen_btn.config(state=tk.NORMAL))
    threading.Thread(target=_task, daemon=True).start()

def run_sync():
    srt_path = clean_path(sync_srt_var.get().strip())
    try: offset_val = float(sync_offset_var.get().strip().replace('+', ''))
    except ValueError: return messagebox.showerror("Error", "Offset must be a number")
    if not srt_path or not os.path.exists(srt_path): return messagebox.showerror("Error", "SRT file not found!")

    def _task():
        try:
            app.after(0, lambda: sync_btn.config(state=tk.DISABLED))
            update_status("Adjusting SRT timestamps...")
            segments = parse_srt(srt_path)
            for seg in segments:
                seg['start'] = max(0, seg['start'] + offset_val)
                seg['end']   = max(0.001, seg['end'] + offset_val)
                if seg['end'] <= seg['start']: seg['end'] = seg['start'] + 0.5

            srt_content = "".join(f"{i+1}\n{format_timestamp(s['start'])} --> {format_timestamp(s['end'])}\n{s['text']}\n\n"
                                  for i, s in enumerate(segments))
            base      = re.sub(r'_synced$', '', os.path.splitext(os.path.basename(srt_path))[0])
            save_path = clean_path(os.path.join(config.SUB_DIR, f"{base}_synced.srt"))
            with open(save_path, 'w', encoding='utf-8') as f: f.write(srt_content)

            update_status(f"✅ Sync done! Offset: {offset_val:+.2f}s")
            update_progress(100, 100, "%")
            populate_all_tabs(save_path)
            
            # Apply subtitle to VLC
            apply_subtitle_to_vlc(app, save_path, f"✅ Sync Applied!\nOffset: {offset_val:+.2f} seconds", clean_path(gen_video_var.get()))
        except Exception as e:
            app.after(0, lambda: messagebox.showerror("Sync Error", str(e)))
            update_status("Error.")
        finally:
            app.after(0, lambda: sync_btn.config(state=tk.NORMAL))
    threading.Thread(target=_task, daemon=True).start()

import keyring
import os
import base64

SERVICE_NAME = "UniversalSubtitleToolkit"
ACCOUNT_NAME = "api_key_secure_storage"

# Fallback path if Windows Vault is blocked
try:
    FALLBACK_FILE = os.path.join(os.getenv('APPDATA'), 'UniversalSubtitles', '.secure_key')
except Exception:
    FALLBACK_FILE = os.path.join(os.getcwd(), '.secure_key')

def save_api_key(api_key):
    """Saves the API key securely."""
    if api_key and api_key.strip():
        try:
            # Try Windows Credential Manager first
            keyring.set_password(SERVICE_NAME, ACCOUNT_NAME, api_key.strip())
        except Exception:
            # Fallback: Base64 obfuscated file in AppData
            os.makedirs(os.path.dirname(FALLBACK_FILE), exist_ok=True)
            with open(FALLBACK_FILE, "wb") as f:
                f.write(base64.b64encode(api_key.strip().encode()))

def load_api_key():
    """Retrieves the API key securely."""
    try:
        key = keyring.get_password(SERVICE_NAME, ACCOUNT_NAME)
        if key: return key
    except Exception:
        pass
    
    # Check fallback if Windows vault failed
    if os.path.exists(FALLBACK_FILE):
        try:
            with open(FALLBACK_FILE, "rb") as f:
                return base64.b64decode(f.read()).decode()
        except Exception:
            pass
            
    return ""

import math
import os

def format_timestamp(seconds):
    seconds = max(0, seconds)
    h  = math.floor(seconds / 3600)
    m  = math.floor((seconds % 3600) / 60)
    s  = math.floor(seconds % 60)
    ms = math.floor((seconds - math.floor(seconds)) * 1000)
    return f"{h:02d}:{m:02d}:{s:02d},{ms:03d}"

def format_ass_timestamp(seconds):
    seconds = max(0, seconds)
    h  = int(seconds // 3600)
    m  = int((seconds % 3600) // 60)
    s  = int(seconds % 60)
    cs = int(round((seconds - int(seconds)) * 100))
    if cs >= 100: cs = 99
    return f"{h}:{m:02d}:{s:02d}.{cs:02d}"

def time_to_sec(t_str):
    t_str = t_str.replace('.', ',')
    if ',' not in t_str: t_str += ',000'
    h, m, s_ms = t_str.split(':')
    s, ms = s_ms.split(',')
    return int(h)*3600 + int(m)*60 + int(s) + int(ms)/1000.0

def clean_path(path):
    if not path: return ""
    return os.path.normpath(path.replace("\\\\", "\\").replace("//", "/"))

def detect_paired_file(filepath, file_type):
    if not filepath: return ""
    base = os.path.splitext(filepath)[0]
    if file_type == "video":
        for ext in ['.mp4', '.mkv', '.avi', '.ts', '.mov', '.wmv']:
            if os.path.exists(base + ext): return base + ext
    elif file_type == "srt":
        if os.path.exists(base + ".srt"): return base + ".srt"
    return ""

def get_available_subtitles(video_path):

def parse_srt(filepath):
    with open(filepath, 'r', encoding='utf-8-sig') as f:
        content = f.read()
    blocks  = re.split(r'\n\s*\n', content.strip())
    parsed  = []
    for block in blocks:
        lines = block.split('\n')
        if len(lines) >= 3:
            times = lines[1].split(' --> ')
            if len(times) == 2:
                try:
                    parsed.append({
                        "index": lines[0].strip(),
                        "start": time_to_sec(times[0].strip()),
                        "end":   time_to_sec(times[1].strip()),
                        "text":  '\n'.join(lines[2:]).strip()
                    })
                except Exception:
                    pass
    return parsed

def hex_to_ass_color(hex_color, alpha=0):
    hx = hex_color.lstrip('#')
    if len(hx) == 3: hx = ''.join(c*2 for c in hx)
    r, g, b = int(hx[0:2], 16), int(hx[2:4], 16), int(hx[4:6], 16)
    return f"&H{alpha:02X}{b:02X}{g:02X}{r:02X}"

def generate_ass_content(segments, style):
    primary   = hex_to_ass_color(style['primary_color'])

    bg_alpha  = 0x60 if style['background'] else 0xFF
    back      = hex_to_ass_color("#000000", alpha=bg_alpha)

    italic    = -1 if style['italic'] else 0

    header = (

        "PlayResX: 512\n"
        "PlayResY: 288\n\n"
        "[V4+ Styles]\n"
        "Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, "
        "OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, "
        "ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, "
        "Alignment, MarginL, MarginR, MarginV, Encoding\n"
        f"Style: Default,{style['font_name']},{style['font_size']},"

        f"{bold},{italic},0,0,100,100,0,0,1,"

def get_ffmpeg_path():
    """Foolproof method to find ffmpeg.exe whether running as script or compiled .exe"""
    import sys, os
    
    # 1. If running as compiled PyInstaller app (onedir mode)
    if getattr(sys, 'frozen', False):
        exe_dir = os.path.dirname(sys.executable)
        path = os.path.join(exe_dir, "ffmpeg.exe")
        if os.path.exists(path): 
            return path
            
    # 2. Check local script directory (Development mode)
    path = os.path.join(os.path.abspath("."), "ffmpeg.exe")
    if os.path.exists(path): 
        return path
        
    # 3. Fallback to system PATH
    return "ffmpeg"

import os

import tkinter.messagebox as messagebox

def get_vlc_path():
    paths = [
        os.path.join(os.environ.get('PROGRAMW6432', 'C:\\Program Files'), 'VideoLAN', 'VLC', 'vlc.exe'),
        os.path.join(os.environ.get('PROGRAMFILES(X86)', 'C:\\Program Files (x86)'), 'VideoLAN', 'VLC', 'vlc.exe'),
        os.path.join(os.environ.get('PROGRAMFILES', 'C:\\Program Files'), 'VideoLAN', 'VLC', 'vlc.exe')
    ]
    for p in paths:
        if os.path.exists(p): return p
    return None

def launch_vlc_preview(video_path, srt_path):
    if not video_path or not os.path.exists(video_path):
        messagebox.showerror("VLC Preview", "No valid video file selected.\nPlease select a video in Tab 1 first!")
        return
    if not srt_path or not os.path.exists(srt_path):
        messagebox.showerror("VLC Preview", f"No valid subtitle file found at:\n{srt_path}")
        return
        
    vlc = get_vlc_path()
    if not vlc:
        messagebox.showerror("VLC Not Found", "Could not locate vlc.exe.\nPlease ensure VLC is installed in the default directory.")
        return
        
    try:
        cf = subprocess.CREATE_NO_WINDOW if os.name == 'nt' else 0
        
        vid = clean_path(video_path)
        sub = clean_path(srt_path)
        
        # Using a raw string guarantees VLC parses the quotes properly on Windows!
        cmd = f'"{vlc}" --one-instance --no-playlist-enqueue "{vid}" --sub-file="{sub}"'
        
        subprocess.Popen(
            cmd,
            stdin=subprocess.DEVNULL,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            creationflags=cf
        )
    except Exception as e:
        messagebox.showerror("Error", f"Could not launch VLC:\n{e}")

def apply_subtitle_to_vlc(app, subtitle_path, success_message, video_path=None):
    def _execute():
        if video_path and os.path.exists(video_path):
            # Launch VLC seamlessly
            launch_vlc_preview(video_path, subtitle_path)
            # Show success prompt AFTER commanding VLC to open
            messagebox.showinfo("✅ Success", f"{success_message}\n\n▶️ Subtitle sent to VLC Player!")
        else:
            try:
                if os.name == 'nt':
                    cf = subprocess.CREATE_NO_WINDOW if os.name == 'nt' else 0
                    cmd = f'explorer /select,"{clean_path(subtitle_path)}"'
                    subprocess.Popen(cmd, stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, creationflags=cf)
            except Exception:
                pass
            messagebox.showinfo("✅ Success", f"{success_message}\n\n📌 TO APPLY IN VLC:\nThe file is highlighted in Explorer.\nDRAG & DROP it into your VLC window!")

    app.after(0, _execute)

def get_vlc_ext_path():
    """Helper to get the exact path where the Lua extension should live."""
    return os.path.join(os.path.expanduser('~'), 'AppData', 'Roaming', 'vlc', 'lua', 'extensions', 'universal_subtitles.lua')

def install_vlc_extension(app):
    lua_source = resource_path("universal_subtitles.lua")
    if not os.path.exists(lua_source):
        return messagebox.showerror("Error", "Could not find universal_subtitles.lua inside the application files.")

    ext_file = get_vlc_ext_path()
    target_dir = os.path.dirname(ext_file)

    try:
        os.makedirs(target_dir, exist_ok=True)
    except Exception:
        pass

    msg = "This will install the auto-load extension directly into VLC's plugin folder.\n\nClick OK to install it automatically."
    if messagebox.askokcancel("Install VLC Extension", msg, parent=app):
        try:
            shutil.copy2(lua_source, ext_file)
            messagebox.showinfo("✅ Installation Complete", f"Extension installed successfully!\n\nPlease completely close and restart VLC Player.", parent=app)
        except Exception as e:
            messagebox.showerror("Installation Failed", f"Could not copy file:\n{str(e)}", parent=app)

            except Exception as e:
                messagebox.showerror("Error", f"Could not remove extension:\n{e}", parent=app)

        const currentScroll = window.scrollY;
        if (currentScroll > 40) {
            header.classList.add('scrolled');
        } else {
            header.classList.remove('scrolled');
        }
        lastScroll = currentScroll;
    }, { passive: true });

    // ── Floating Particles ──
    const particlesContainer = document.getElementById('particles');
    if (particlesContainer) {
        const particleCount = 25;
        for (let i = 0; i < particleCount; i++) {
            const particle = document.createElement('div');
            particle.classList.add('particle');
            particle.style.left = Math.random() * 100 + '%';
            particle.style.top = (Math.random() * 100 + 100) + '%';
            particle.style.animationDuration = (8 + Math.random() * 15) + 's';
            particle.style.animationDelay = (Math.random() * 10) + 's';
            particle.style.width = (1 + Math.random() * 2) + 'px';
            particle.style.height = particle.style.width;
            particlesContainer.appendChild(particle);
        }
    }

    // ── Mobile Menu Toggle ──
    const mobileBtn = document.getElementById('mobile-menu-btn');
    const mainNav = document.getElementById('main-nav');

    if (mobileBtn && mainNav) {
        mobileBtn.addEventListener('click', () => {
            mainNav.classList.toggle('mobile-open');
            mobileBtn.classList.toggle('active');
        });
    }

    // ── Smooth Scroll for Anchor Links ──
    document.querySelectorAll('a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', (e) => {
            const targetId = anchor.getAttribute('href');
            if (targetId === '#') return;
            const target = document.querySelector(targetId);
            if (target) {
                e.preventDefault();
                target.scrollIntoView({ behavior: 'smooth', block: 'start' });
                // Close mobile nav if open
                if (mainNav) mainNav.classList.remove('mobile-open');
                if (mobileBtn) mobileBtn.classList.remove('active');
            }
        });
    });

    // ── Subtle Parallax on Hero Showcase ──
    const heroFrame = document.querySelector('.hero-showcase-frame');
    if (heroFrame && window.innerWidth > 960) {
        window.addEventListener('mousemove', (e) => {
            const x = (e.clientX / window.innerWidth - 0.5) * 8;
            const y = (e.clientY / window.innerHeight - 0.5) * 8;
            heroFrame.style.transform = `perspective(1000px) rotateY(${x}deg) rotateX(${-y}deg)`;
        }, { passive: true });
    }

});
